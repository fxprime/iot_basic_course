# บทที่ 7.5: LCD 1602 Display - แสดงผลข้อมูล

ในบทนี้เราจะเรียนรู้การใช้งาน LCD 1602 (16 ตัวอักษร x 2 แถว) ผ่าน I2C เพื่อแสดงผลข้อมูลต่างๆ

## วัตถุประสงค์
- เข้าใจการทำงานของ LCD 1602
- ใช้ I2C สื่อสารกับ LCD
- แสดงผลข้อความและตัวเลข
- สร้างอักขระพิเศษ (Custom Characters)

## ขั้นตอนที่ 1: รู้จักกับ LCD 1602

### 1.1 LCD 1602 คืออะไร?

**LCD 1602** = จอแสดงผลข้อความ 16x2 (16 ตัวอักษร × 2 แถว)

![รูป LCD 1602](../assets/images/lcd1602.png)

**ลักษณะเด่น:**
- แสดงได้ 32 ตัวอักษร (16×2)
- มี backlight (ไฟพื้นหลัง)
- ใช้ I2C module (4 สาย: VCC, GND, SDA, SCL)
- ราคาถูก ใช้งานง่าย

<details markdown="1">
<summary>📖 <b>I2C vs Parallel Interface</b></summary>

### LCD 1602 มี 2 แบบ:

**1. Parallel (16 pins):**
```t
- ต้องใช้ 6-8 GPIO pins
- ต่อวงจรซับซ้อน
- ควบคุมตรง แต่เสียพอร์ต
```

**2. I2C Module (4 pins):**
```
- ใช้แค่ 2 GPIO pins (SDA, SCL)
- ต่อง่าย แชร์ bus ได้
- ใช้ I2C address (โดยทั่วไป 0x27 หรือ 0x3F)
```

**เราจะใช้แบบ I2C!** 🎉

### I2C Protocol

**I2C (Inter-Integrated Circuit):**
- Serial communication protocol
- ใช้ 2 สาย: SDA (Data), SCL (Clock)
- หลาย device ใช้ bus เดียวกันได้ (ต่าง address)
- ความเร็ว: 100kHz (Standard) หรือ 400kHz (Fast)

</details>

### 1.2 การต่อวงจร

```
LCD I2C Module    ESP32
--------------    ------
VCC           →   5V (หรือ 3.3V*)
GND           →   GND
SDA           →   GPIO 21 (SDA)
SCL           →   GPIO 22 (SCL)
```

![รูปต่อวงจร LCD I2C](../assets/images/lcd-i2c-wiring.png)

⚠️ **หมายเหตุ:**
- ส่วนใหญ่ LCD I2C module รองรับทั้ง 3.3V และ 5V
- ถ้าใช้ 3.3V backlight อาจสว่างน้อยกว่า
- ESP32 มี internal pull-up สำหรับ I2C อยู่แล้ว

4. คลิก **Install**

![รูปติดตั้ง Library](../assets/images/lcd-library-install.png)

### 2.2 หา I2C Address

ก่อนใช้งาน ต้องรู้ I2C address ของ LCD ก่อน:

```cpp
#include <Wire.h>

void setup() {
  Serial.begin(115200);
  Wire.begin();
  
  Serial.println("I2C Scanner");
  Serial.println("Scanning...");
  
  byte count = 0;
  for (byte i = 1; i < 127; i++) {
    Wire.beginTransmission(i);
    if (Wire.endTransmission() == 0) {
      Serial.print("Found device at address: 0x");
      if (i < 16) Serial.print("0");
      Serial.println(i, HEX);
      count++;
    }
  }
  
  Serial.print("Found ");
  Serial.print(count);
  Serial.println(" device(s)");
}

void loop() {}
```

**ผลลัพธ์:**
```
I2C Scanner
Scanning...
Found device at address: 0x27
Found 1 device(s)
```

✅ จดaddress นี้ไว้ (มักจะเป็น `0x27` หรือ `0x3F`)

## ขั้นตอนที่ 3: โปรแกรมพื้นฐาน

### 3.1 Hello World!

```cpp
#include <Wire.h>
#include <LiquidCrystal_I2C.h>

// สร้าง object (address, columns, rows)
LiquidCrystal_I2C lcd(0x27, 16, 2);

void setup() {
  // เริ่มต้น LCD
  lcd.init();
  
  // เปิด backlight
  lcd.backlight();
  
  // แสดงข้อความ
  lcd.setCursor(0, 0);  // คอลัมน์ 0, แถว 0
  lcd.print("Hello, World!");
  
  lcd.setCursor(0, 1);  // คอลัมน์ 0, แถว 1
  lcd.print("ESP32 + LCD!");
}

void loop() {
  // ไม่ต้องทำอะไร
}
```

**ผลลัพธ์บน LCD:**
```
Hello, World!   
ESP32 + LCD!    
```

### LiquidCrystal_I2C Functions

**1. การเริ่มต้น:**
```cpp
lcd.init();          // เริ่มต้น LCD
lcd.backlight();     // เปิด backlight
lcd.noBacklight();   // ปิด backlight
```

**2. การเคลียร์:**
```cpp
lcd.clear();         // ลบทั้งหมด
lcd.home();          // กลับไปตำแหน่ง (0,0)
```

**3. การตั้งตำแหน่ง:**
```cpp
lcd.setCursor(col, row);  // col: 0-15, row: 0-1
lcd.setCursor(0, 0);      // บรรทัดบน ซ้ายสุด
lcd.setCursor(0, 1);      // บรรทัดล่าง ซ้ายสุด
lcd.setCursor(10, 0);     // บรรทัดบน ตัวที่ 11
```

**4. การแสดงผล:**
```cpp
lcd.print("Text");        // แสดงข้อความ
lcd.print(123);           // แสดงตัวเลข
lcd.print(3.14);          // แสดงทศนิยม
lcd.print(value, DEC);    // ฐาน 10
lcd.print(value, HEX);    // ฐาน 16
lcd.print(value, BIN);    // ฐาน 2
```

**5. Cursor & Display:**
```cpp
lcd.cursor();            // แสดง cursor (ขีดล่าง)
lcd.noCursor();          // ซ่อน cursor
lcd.blink();             // cursor กระพริบ
lcd.noBlink();           // cursor ไม่กระพริบ
lcd.display();           // เปิดแสดงผล
lcd.noDisplay();         // ปิดแสดงผล (ข้อมูลยังอยู่)
```

**6. Scrolling:**
```cpp
lcd.scrollDisplayLeft();  // เลื่อนซ้าย
lcd.scrollDisplayRight(); // เลื่อนขวา
lcd.autoscroll();         // เลื่อนอัตโนมัติ
lcd.noAutoscroll();       // ไม่เลื่อนอัตโนมัติ
```

</details>

### 3.2 แสดงตัวเลขและอัพเดท

```cpp
#include <Wire.h>
#include <LiquidCrystal_I2C.h>

LiquidCrystal_I2C lcd(0x27, 16, 2);

int counter = 0;

void setup() {
  lcd.init();
  lcd.backlight();
  
  lcd.setCursor(0, 0);
  lcd.print("Counter:");
}

void loop() {
  // อัพเดทเฉพาะส่วนตัวเลข
  lcd.setCursor(9, 0);
  lcd.print(counter);
  lcd.print("   ");  // ล้างตัวเลขเก่า (ถ้าสั้นลง)
  
  // แสดง progress bar
  lcd.setCursor(0, 1);
  int bars = counter % 17;  // 0-16
  for (int i = 0; i < bars; i++) {
    lcd.print("=");
  }
  for (int i = bars; i < 16; i++) {
    lcd.print(" ");
  }
  
  counter++;
  delay(500);
}
```

**ผลลัพธ์:**
```
Counter: 42     
================
```

## ขั้นตอนที่ 4: โปรเจคขั้นสูง

### 4.1 นาฬิกา Uptime

```cpp
#include <Wire.h>
#include <LiquidCrystal_I2C.h>

LiquidCrystal_I2C lcd(0x27, 16, 2);

void setup() {
  lcd.init();
  lcd.backlight();
  
  lcd.setCursor(0, 0);
  lcd.print("Uptime:");
}

void loop() {
  unsigned long totalSeconds = millis() / 1000;
  
  int hours = totalSeconds / 3600;
  int minutes = (totalSeconds % 3600) / 60;
  int seconds = totalSeconds % 60;
  
  lcd.setCursor(0, 1);
  
  // แสดง HH:MM:SS
  if (hours < 10) lcd.print("0");
  lcd.print(hours);
  lcd.print(":");
  
  if (minutes < 10) lcd.print("0");
  lcd.print(minutes);
  lcd.print(":");
  
  if (seconds < 10) lcd.print("0");
  lcd.print(seconds);
  
  delay(1000);
}
```

### 4.2 แสดงข้อมูลเซ็นเซอร์

```cpp
#include <Wire.h>
#include <LiquidCrystal_I2C.h>

LiquidCrystal_I2C lcd(0x27, 16, 2);

const int SENSOR_PIN = 34;

void setup() {
  lcd.init();
  lcd.backlight();
  pinMode(SENSOR_PIN, INPUT);
  
  lcd.setCursor(0, 0);
  lcd.print("Sensor Reading");
}

void loop() {
  int rawValue = analogRead(SENSOR_PIN);
  float voltage = rawValue * (3.3 / 4095.0);
  
  // แสดงค่า raw
  lcd.setCursor(0, 1);
  lcd.print("ADC:");
  lcd.print(rawValue);
  lcd.print("   ");
  
  // แสดงแรงดัน
  lcd.setCursor(9, 1);
  lcd.print(voltage, 2);
  lcd.print("V");
  
  delay(500);
}
```

### 4.3 สร้างอักขระพิเศษ (Custom Characters)

```cpp
#include <Wire.h>
#include <LiquidCrystal_I2C.h>

LiquidCrystal_I2C lcd(0x27, 16, 2);

// กำหนดรูปแบบ (8x5 pixels)
byte heart[] = {
  0b00000,
  0b01010,
  0b11111,
  0b11111,
  0b11111,
  0b01110,
  0b00100,
  0b00000
};

byte smiley[] = {
  0b00000,
  0b01010,
  0b01010,
  0b00000,
  0b10001,
  0b01110,
  0b00000,
  0b00000
};

byte degree[] = {
  0b00110,
  0b01001,
  0b01001,
  0b00110,
  0b00000,
  0b00000,
  0b00000,
  0b00000
};

byte arrowUp[] = {
  0b00100,
  0b01110,
  0b11111,
  0b00100,
  0b00100,
  0b00100,
  0b00100,
  0b00000
};

byte arrowDown[] = {
  0b00100,
  0b00100,
  0b00100,
  0b00100,
  0b11111,
  0b01110,
  0b00100,
  0b00000
};

void setup() {
  lcd.init();
  lcd.backlight();
  
  // สร้างอักขระพิเศษ (0-7)
  lcd.createChar(0, heart);
  lcd.createChar(1, smiley);
  lcd.createChar(2, degree);
  lcd.createChar(3, arrowUp);
  lcd.createChar(4, arrowDown);
  
  // แสดงผล
  lcd.setCursor(0, 0);
  lcd.print("Custom Chars:");
  
  lcd.setCursor(0, 1);
  lcd.write(0);  // heart
  lcd.print(" ");
  lcd.write(1);  // smiley
  lcd.print(" Temp:25");
  lcd.write(2);  // degree
  lcd.print("C ");
  lcd.write(3);  // arrow up
}

void loop() {
  // กระพริบลูกศร
  lcd.setCursor(15, 1);
  lcd.write(3);  // up
  delay(500);
  lcd.setCursor(15, 1);
  lcd.write(4);  // down
  delay(500);
}
```

**ผลลัพธ์:**
```
Custom Chars:   
♥ ☺ Temp:25°C ↑
```

### 4.4 Menu System

```cpp
#include <Wire.h>
#include <LiquidCrystal_I2C.h>

LiquidCrystal_I2C lcd(0x27, 16, 2);

const int BUTTON_UP = 18;
const int BUTTON_DOWN = 19;
const int BUTTON_SELECT = 23;

int menuIndex = 0;
const int MENU_ITEMS = 4;

String menuItems[] = {
  "1. LED Control",
  "2. Servo Test",
  "3. Sensor Read",
  "4. Settings"
};

void setup() {
  lcd.init();
  lcd.backlight();
  
  pinMode(BUTTON_UP, INPUT_PULLUP);
  pinMode(BUTTON_DOWN, INPUT_PULLUP);
  pinMode(BUTTON_SELECT, INPUT_PULLUP);
  
  displayMenu();
}

void displayMenu() {
  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("> ");
  lcd.print(menuItems[menuIndex]);
  
  // แสดงรายการถัดไป (ถ้ามี)
  if (menuIndex + 1 < MENU_ITEMS) {
    lcd.setCursor(2, 1);
    lcd.print(menuItems[menuIndex + 1]);
  }
}

void loop() {
  // ปุ่มขึ้น
  if (digitalRead(BUTTON_UP) == LOW) {
    if (menuIndex > 0) {
      menuIndex--;
      displayMenu();
    }
    delay(200);
  }
  
  // ปุ่มลง
  if (digitalRead(BUTTON_DOWN) == LOW) {
    if (menuIndex < MENU_ITEMS - 1) {
      menuIndex++;
      displayMenu();
    }
    delay(200);
  }
  
  // ปุ่มเลือก
  if (digitalRead(BUTTON_SELECT) == LOW) {
    lcd.clear();
    lcd.setCursor(0, 0);
    lcd.print("Selected:");
    lcd.setCursor(0, 1);
    lcd.print(menuItems[menuIndex]);
    delay(2000);
    displayMenu();
  }
}
```
