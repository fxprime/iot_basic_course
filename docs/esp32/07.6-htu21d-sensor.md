# บทที่ 7.6: HTU21D - Temperature & Humidity Sensor

ในบทนี้เราจะเรียนรู้การใช้งานเซ็นเซอร์ HTU21D วัดอุณหภูมิและความชื้น แล้วแสดงผลบน LCD

## วัตถุประสงค์
- เข้าใจการทำงานของ HTU21D sensor
- ใช้ I2C อ่านค่าจากเซ็นเซอร์
- แสดงผลบน LCD 1602
- สร้างระบบเตือนอุณหภูมิ/ความชื้น

---

## ขั้นตอนที่ 1: รู้จักกับ HTU21D

### 1.1 HTU21D คืออะไร?

**HTU21D** = เซ็นเซอร์วัดอุณหภูมิและความชื้นความแม่นยำสูง

![รูป HTU21D](../assets/images/htu21d-sensor.png)

**ลักษณะเด่น:**
- วัดอุณหภูมิ: -40°C ถึง +125°C (±0.3°C)
- วัดความชื้น: 0-100% RH (±2%)
- ใช้ I2C communication (address: 0x40)
- แรงดัน: 2.1-3.6V (เหมาะกับ 3.3V)
- กินไฟต่ำมาก

<details markdown="1">
<summary>📖 <b>HTU21D vs DHT11/DHT22</b></summary>

### เปรียบเทียบเซ็นเซอร์

| Feature | HTU21D | DHT22 | DHT11 |
|---------|--------|-------|-------|
| **อุณหภูมิ** | ±0.3°C | ±0.5°C | ±2°C |
| **ความชื้น** | ±2% | ±2-5% | ±5% |
| **Interface** | I2C | 1-Wire | 1-Wire |
| **Sampling** | Fast | 2s | 1s |
| **ราคา** | แพงกว่า | กลาง | ถูกสุด |
| **ขนาด** | เล็ก | กลาง | ใหญ่ |

**HTU21D เหมาะกับ:**
- โปรเจคต้องการความแม่นยำสูง
- ใช้ I2C bus แชร์กับอุปกรณ์อื่น
- ต้องการ sampling เร็ว
- ประหยัดไฟ (battery powered)

</details>

### 1.2 การต่อวงจร

```
HTU21D Module     ESP32
-------------     ------
VCC           →   3.3V
GND           →   GND
SDA           →   GPIO 21 (SDA)
SCL           →   GPIO 22 (SCL)
```

![รูปต่อวงจร HTU21D](../assets/images/htu21d-wiring.png)

**หมายเหตุ:**
- HTU21D module มักมี pull-up resistor ในตัวแล้ว
- สามารถต่อร่วม bus กับ LCD I2C ได้

## ขั้นตอนที่ 2: ติดตั้ง Library

### 2.1 ติดตั้ง SparkFun HTU21D Library

**Arduino IDE:**
1. **Sketch** → **Include Library** → **Manage Libraries...**
2. ค้นหา: `SparkFun HTU21D`
3. เลือก `SparkFun HTU21D Humidity and Temperature Sensor Breakout`
4. คลิก **Install**

### 2.2 ทดสอบการเชื่อมต่อ

```cpp
#include <Wire.h>
#include "SparkFunHTU21D.h"

HTU21D htu;

void setup() {
  Serial.begin(115200);
  Wire.begin();
  
  Serial.println("HTU21D Test");
  
  // เริ่มต้นเซ็นเซอร์
  if (htu.begin() == false) {
    Serial.println("❌ HTU21D not detected!");
    while (1);  // หยุด
  }
  
  Serial.println("✓ HTU21D detected!");
}

void loop() {
  float humidity = htu.readHumidity();
  float temp = htu.readTemperature();
  
  Serial.print("Temperature: ");
  Serial.print(temp, 1);
  Serial.print("°C  |  Humidity: ");
  Serial.print(humidity, 1);
  Serial.println("%");
  
  delay(2000);
}
```

**ผลลัพธ์:**
```
HTU21D Test
✓ HTU21D detected!
Temperature: 25.3°C  |  Humidity: 62.5%
Temperature: 25.4°C  |  Humidity: 62.3%
```

## ขั้นตอนที่ 3: แสดงผลบน LCD

### 3.1 โปรแกรมพื้นฐาน

```cpp
#include <Wire.h>
#include <LiquidCrystal_I2C.h>
#include "SparkFunHTU21D.h"

// LCD (address 0x27, 16x2)
LiquidCrystal_I2C lcd(0x27, 16, 2);

// HTU21D sensor
HTU21D htu;

// Custom character: degree symbol
byte degreeChar[] = {
  0b00110,
  0b01001,
  0b01001,
  0b00110,
  0b00000,
  0b00000,
  0b00000,
  0b00000
};

void setup() {
  Serial.begin(115200);
  Wire.begin();
  
  // เริ่มต้น LCD
  lcd.init();
  lcd.backlight();
  lcd.createChar(0, degreeChar);
  
  // เริ่มต้น HTU21D
  if (htu.begin() == false) {
    lcd.setCursor(0, 0);
    lcd.print("HTU21D Error!");
    Serial.println("❌ HTU21D not detected!");
    while (1);
  }
  
  lcd.setCursor(0, 0);
  lcd.print("HTU21D Ready!");
  delay(2000);
  lcd.clear();
}

void loop() {
  // อ่านค่าจากเซ็นเซอร์
  float temperature = htu.readTemperature();
  float humidity = htu.readHumidity();
  
  // แสดงอุณหภูมิ
  lcd.setCursor(0, 0);
  lcd.print("Temp: ");
  lcd.print(temperature, 1);
  lcd.write(0);  // degree symbol
  lcd.print("C  ");
  
  // แสดงความชื้น
  lcd.setCursor(0, 1);
  lcd.print("Humi: ");
  lcd.print(humidity, 1);
  lcd.print("%   ");
  
  // Serial Monitor
  Serial.print("Temp: ");
  Serial.print(temperature, 1);
  Serial.print("°C  |  Humi: ");
  Serial.print(humidity, 1);
  Serial.println("%");
  
  delay(2000);
}
```

**ผลลัพธ์บน LCD:**
```
Temp: 25.3°C    
Humi: 62.5%     
```

### 3.2 เพิ่มกราฟแท่ง (Bar Graph)

```cpp
#include <Wire.h>
#include <LiquidCrystal_I2C.h>
#include "SparkFunHTU21D.h"

LiquidCrystal_I2C lcd(0x27, 16, 2);
HTU21D htu;

// Custom characters for bar graph
byte bar1[] = {0b10000, 0b10000, 0b10000, 0b10000, 0b10000, 0b10000, 0b10000, 0b10000};
byte bar2[] = {0b11000, 0b11000, 0b11000, 0b11000, 0b11000, 0b11000, 0b11000, 0b11000};
byte bar3[] = {0b11100, 0b11100, 0b11100, 0b11100, 0b11100, 0b11100, 0b11100, 0b11100};
byte bar4[] = {0b11110, 0b11110, 0b11110, 0b11110, 0b11110, 0b11110, 0b11110, 0b11110};
byte bar5[] = {0b11111, 0b11111, 0b11111, 0b11111, 0b11111, 0b11111, 0b11111, 0b11111};
byte degreeChar[] = {0b00110, 0b01001, 0b01001, 0b00110, 0b00000, 0b00000, 0b00000, 0b00000};

void setup() {
  Serial.begin(115200);
  Wire.begin();
  
  lcd.init();
  lcd.backlight();
  
  // สร้าง custom characters
  lcd.createChar(0, degreeChar);
  lcd.createChar(1, bar1);
  lcd.createChar(2, bar2);
  lcd.createChar(3, bar3);
  lcd.createChar(4, bar4);
  lcd.createChar(5, bar5);
  
  if (htu.begin() == false) {
    lcd.print("HTU21D Error!");
    while (1);
  }
  
  lcd.print("HTU21D Ready!");
  delay(2000);
  lcd.clear();
}

void displayBar(int col, int row, int value, int maxValue) {
  // คำนวณจำนวน bars (0-5)
  int level = map(value, 0, maxValue, 0, 5);
  level = constrain(level, 0, 5);
  
  lcd.setCursor(col, row);
  if (level == 0) {
    lcd.print(" ");
  } else {
    lcd.write(level);  // แสดง bar character
  }
}

void loop() {
  float temperature = htu.readTemperature();
  float humidity = htu.readHumidity();
  
  // แสดงอุณหภูมิพร้อมกราฟ
  lcd.setCursor(0, 0);
  lcd.print("T:");
  lcd.print(temperature, 1);
  lcd.write(0);  // degree
  lcd.print("C ");
  
  // กราฟอุณหภูมิ (0-50°C)
  int tempBars = map(temperature, 0, 50, 0, 50);
  for (int i = 0; i < 10; i++) {
    int barValue = constrain(tempBars - (i * 5), 0, 5);
    lcd.setCursor(i + 6, 0);
    if (barValue == 0) {
      lcd.print(" ");
    } else {
      lcd.write(barValue);
    }
  }
  
  // แสดงความชื้นพร้อมกราฟ
  lcd.setCursor(0, 1);
  lcd.print("H:");
  lcd.print(humidity, 1);
  lcd.print("% ");
  
  // กราฟความชื้น (0-100%)
  int humiBars = map(humidity, 0, 100, 0, 50);
  for (int i = 0; i < 10; i++) {
    int barValue = constrain(humiBars - (i * 5), 0, 5);
    lcd.setCursor(i + 6, 1);
    if (barValue == 0) {
      lcd.print(" ");
    } else {
      lcd.write(barValue);
    }
  }
  
  Serial.print("T: ");
  Serial.print(temperature, 1);
  Serial.print("°C  |  H: ");
  Serial.print(humidity, 1);
  Serial.println("%");
  
  delay(2000);
}
```

**ผลลัพธ์บน LCD:**
```
T:25.3°C ▓▓▓▓▓░░░░░
H:62.5% ▓▓▓▓▓▓▓░░░
```

## ขั้นตอนที่ 4: ระบบเตือน (Alert System)

### 4.1 เตือนเมื่อค่าผิดปกติ

```cpp
#include <Wire.h>
#include <LiquidCrystal_I2C.h>
#include "SparkFunHTU21D.h"

LiquidCrystal_I2C lcd(0x27, 16, 2);
HTU21D htu;

// Threshold values
const float TEMP_MAX = 30.0;
const float TEMP_MIN = 20.0;
const float HUMI_MAX = 70.0;
const float HUMI_MIN = 40.0;

const int LED_RED = 25;
const int LED_GREEN = 26;
const int BUZZER_PIN = 27;

byte degreeChar[] = {0b00110, 0b01001, 0b01001, 0b00110, 0b00000, 0b00000, 0b00000, 0b00000};
byte alertChar[] = {0b00100, 0b01110, 0b01110, 0b01110, 0b00100, 0b00000, 0b00100, 0b00000};

void setup() {
  Serial.begin(115200);
  Wire.begin();
  
  lcd.init();
  lcd.backlight();
  lcd.createChar(0, degreeChar);
  lcd.createChar(1, alertChar);
  
  pinMode(LED_RED, OUTPUT);
  pinMode(LED_GREEN, OUTPUT);
  pinMode(BUZZER_PIN, OUTPUT);
  
  if (htu.begin() == false) {
    lcd.print("HTU21D Error!");
    while (1);
  }
  
  lcd.print("Climate Monitor");
  lcd.setCursor(0, 1);
  lcd.print("Initializing...");
  delay(2000);
  lcd.clear();
}

void checkAlert(float temp, float humi) {
  bool alert = false;
  String alertMsg = "";
  
  // ตรวจสอบอุณหภูมิ
  if (temp > TEMP_MAX) {
    alert = true;
    alertMsg = "Temp HIGH!";
  } else if (temp < TEMP_MIN) {
    alert = true;
    alertMsg = "Temp LOW!";
  }
  
  // ตรวจสอบความชื้น
  if (humi > HUMI_MAX) {
    alert = true;
    if (alertMsg.length() > 0) alertMsg = "T+H Alert!";
    else alertMsg = "Humi HIGH!";
  } else if (humi < HUMI_MIN) {
    alert = true;
    if (alertMsg.length() > 0) alertMsg = "T+H Alert!";
    else alertMsg = "Humi LOW!";
  }
  
  if (alert) {
    // เปิด LED แดง และ buzzer
    digitalWrite(LED_RED, HIGH);
    digitalWrite(LED_GREEN, LOW);
    tone(BUZZER_PIN, 1000, 200);
    
    // แสดงข้อความเตือนบน LCD (กระพริบ)
    lcd.setCursor(0, 1);
    lcd.write(1);  // alert icon
    lcd.print(" ");
    lcd.print(alertMsg);
    lcd.print("  ");
    
    Serial.print("⚠️  ALERT: ");
    Serial.println(alertMsg);
  } else {
    // สภาวะปกติ
    digitalWrite(LED_RED, LOW);
    digitalWrite(LED_GREEN, HIGH);
    noTone(BUZZER_PIN);
    
    lcd.setCursor(0, 1);
    lcd.print("Status: OK     ");
  }
}

void loop() {
  float temperature = htu.readTemperature();
  float humidity = htu.readHumidity();
  
  // แสดงค่า
  lcd.setCursor(0, 0);
  lcd.print("T:");
  lcd.print(temperature, 1);
  lcd.write(0);
  lcd.print(" H:");
  lcd.print(humidity, 1);
  lcd.print("%  ");
  
  // ตรวจสอบ alert
  checkAlert(temperature, humidity);
  
  Serial.print("T: ");
  Serial.print(temperature, 1);
  Serial.print("°C  |  H: ");
  Serial.print(humidity, 1);
  Serial.println("%");
  
  delay(2000);
}
```

## ขั้นตอนที่ 5: โปรเจคขั้นสูง

### 5.1 Data Logger พร้อม Timestamp

```cpp
#include <Wire.h>
#include <LiquidCrystal_I2C.h>
#include "SparkFunHTU21D.h"

LiquidCrystal_I2C lcd(0x27, 16, 2);
HTU21D htu;

struct DataLog {
  unsigned long timestamp;
  float temperature;
  float humidity;
};

const int LOG_SIZE = 100;
DataLog logs[LOG_SIZE];
int logIndex = 0;
int logCount = 0;

void setup() {
  Serial.begin(115200);
  Wire.begin();
  
  lcd.init();
  lcd.backlight();
  
  if (htu.begin() == false) {
    lcd.print("HTU21D Error!");
    while (1);
  }
  
  Serial.println("Data Logger Started");
  Serial.println("Timestamp,Temperature,Humidity");
}

void logData(float temp, float humi) {
  logs[logIndex].timestamp = millis() / 1000;  // seconds
  logs[logIndex].temperature = temp;
  logs[logIndex].humidity = humi;
  
  // Serial CSV output
  Serial.print(logs[logIndex].timestamp);
  Serial.print(",");
  Serial.print(temp, 2);
  Serial.print(",");
  Serial.println(humi, 2);
  
  logIndex = (logIndex + 1) % LOG_SIZE;
  if (logCount < LOG_SIZE) logCount++;
}

void displayStats() {
  if (logCount == 0) return;
  
  float avgTemp = 0;
  float avgHumi = 0;
  
  for (int i = 0; i < logCount; i++) {
    avgTemp += logs[i].temperature;
    avgHumi += logs[i].humidity;
  }
  
  avgTemp /= logCount;
  avgHumi /= logCount;
  
  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("Avg T:");
  lcd.print(avgTemp, 1);
  lcd.print((char)223);  // degree
  lcd.print("C");
  
  lcd.setCursor(0, 1);
  lcd.print("Avg H:");
  lcd.print(avgHumi, 1);
  lcd.print("%");
  
  delay(3000);
  lcd.clear();
}

void loop() {
  static unsigned long lastLog = 0;
  static unsigned long lastDisplay = 0;
  
  unsigned long now = millis();
  
  // Log every 10 seconds
  if (now - lastLog >= 10000) {
    float temp = htu.readTemperature();
    float humi = htu.readHumidity();
    logData(temp, humi);
    lastLog = now;
  }
  
  // Update display every 2 seconds
  if (now - lastDisplay >= 2000) {
    float temp = htu.readTemperature();
    float humi = htu.readHumidity();
    
    lcd.setCursor(0, 0);
    lcd.print("T:");
    lcd.print(temp, 1);
    lcd.print((char)223);
    lcd.print("C H:");
    lcd.print(humi, 1);
    lcd.print("%");
    
    lcd.setCursor(0, 1);
    lcd.print("Logs: ");
    lcd.print(logCount);
    lcd.print("/");
    lcd.print(LOG_SIZE);
    lcd.print("   ");
    
    lastDisplay = now;
  }
  
  // Check for stats command
  if (Serial.available() > 0) {
    String cmd = Serial.readStringUntil('\n');
    cmd.trim();
    if (cmd == "STATS") {
      displayStats();
    }
  }
}
```
