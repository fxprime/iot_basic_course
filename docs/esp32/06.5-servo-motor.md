# บทที่ 6.5: Servo Motor SG90 - การควบคุมมุมด้วย PWM

ในบทนี้เราจะเรียนรู้การควบคุม Servo Motor แบบ manual ด้วย PWM timing และใช้ Serial Command เพื่อควบคุมมุมแบบเรียลไทม์

## วัตถุประสงค์
- เข้าใจหลักการทำงานของ Servo Motor
- ใช้ PWM timing แบบ manual (microseconds)
- ควบคุมมุมด้วย Serial Command
- เข้าใจความสำคัญของ pulse width

---

## ขั้นตอนที่ 1: รู้จักกับ Servo Motor

### 1.1 Servo Motor คืออะไร?

**Servo Motor** = มอเตอร์ที่สามารถควบคุม**มุม**ได้แม่นยำ

![รูป SG90 Servo](../assets/images/sg90-servo.png)

**ลักษณะเด่น:**
- ควบคุมได้ 0° ถึง 180° (SG90)
- ใช้สัญญาณ PWM ควบคุม
- มี 3 สาย: สีน้ำตาล (GND), สีแดง (VCC 5V), สีส้ม/เหลือง (Signal)

<details markdown="1">
<summary>📖 <b>การทำงานของ Servo Motor</b></summary>

### Servo vs DC Motor

| Servo Motor | DC Motor |
|-------------|----------|
| ควบคุม**มุม** | ควบคุม**ความเร็ว** |
| 0-180° | หมุนต่อเนื่อง |
| ใช้ PWM Signal | ใช้แรงดัน |
| มี feedback circuit | ไม่มี feedback |

### PWM Signal สำหรับ Servo

```
Period = 20ms (50Hz)

Pulse Width:
- 500μs  = 0°   (ซ้ายสุด)
- 1500μs = 90°  (กึ่งกลาง)
- 2500μs = 180° (ขวาสุด)
```

**ตัวอย่าง:**
```
     ┌─┐
     │ │               
─────┘ └────────────── (500μs = 0°)

     ┌────┐
     │    │            
─────┘    └────────── (1500μs = 90°)

     ┌──────┐
     │      │          
─────┘      └──────── (2500μs = 180°)
```

### สูตรคำนวณ

```
Pulse Width (μs) = 500 + (angle × 2000 / 180)

ตัวอย่าง:
- 0°   → 500 + (0 × 2000/180)   = 500μs
- 45°  → 500 + (45 × 2000/180)  = 1000μs
- 90°  → 500 + (90 × 2000/180)  = 1500μs
- 135° → 500 + (135 × 2000/180) = 2000μs
- 180° → 500 + (180 × 2000/180) = 2500μs
```

</details>

### 1.2 การต่อวงจร

```
SG90 Servo       ESP32
-----------      ------
Brown (GND) →    GND
Red (VCC)   →    5V (หรือ external power)
Orange      →    GPIO 25 (Signal)
```

![รูปต่อวงจร Servo](../assets/images/servo-wiring.png)

⚠️ **หมายเหตุ:**
- Servo ใช้กระแสมาก (peak ~500mA) แนะนำใช้ external power 5V
- ถ้าใช้ servo เดียวและไม่มีน้ำหนัก สามารถใช้ 5V จาก ESP32 ได้

## ขั้นตอนที่ 2: ควบคุม Servo แบบ Manual (ไม่ใช้ Library)

### 2.1 สร้าง PWM Signal เอง

```cpp
const int SERVO_PIN = 25;

// PWM Configuration
const int PWM_FREQ = 50;        // 50Hz (20ms period)
const int PWM_RESOLUTION = 16;  // 16-bit resolution (0-65535)

// คำนวณค่า duty cycle
const int PWM_MAX = 65535;  // 2^16 - 1

void setup() {
  Serial.begin(115200);
  
  // ตั้งค่า PWM
  ledcAttach(SERVO_PIN, PWM_FREQ, PWM_RESOLUTION);
  
  Serial.println("=== Servo Control (Manual PWM) ===");
  Serial.println("Initializing...");
}

// ฟังก์ชันแปลง microseconds เป็น duty cycle
int usToDuty(int microseconds) {
  // Period = 20ms = 20000μs
  // duty = (microseconds / 20000) * PWM_MAX
  return (microseconds * PWM_MAX) / 20000;
}

// ฟังก์ชันควบคุม Servo ด้วย pulse width (μs)
void servoWriteUs(int microseconds) {
  // จำกัดค่าให้อยู่ในช่วง 500-2500μs
  microseconds = constrain(microseconds, 500, 2500);
  
  int duty = usToDuty(microseconds);
  ledcWrite(SERVO_PIN, duty);
  
  Serial.print("Pulse: ");
  Serial.print(microseconds);
  Serial.println(" μs");
}

void loop() {
  // ทดสอบหมุนจาก 0° ถึง 180°
  
  Serial.println("Moving to 0°");
  servoWriteUs(500);   // 0°
  delay(1000);
  
  Serial.println("Moving to 90°");
  servoWriteUs(1500);  // 90°
  delay(1000);
  
  Serial.println("Moving to 180°");
  servoWriteUs(2500);  // 180°
  delay(1000);
  
  Serial.println("Moving to 90°");
  servoWriteUs(1500);  // 90°
  delay(1000);
}
```

<details markdown="1">
<summary>💡 <b>อธิบายโค้ด</b></summary>

**1. PWM Configuration**
```cpp
const int PWM_FREQ = 50;        // 50Hz = Period 20ms
const int PWM_RESOLUTION = 16;  // 16-bit = 0-65535
```

**2. การคำนวณ Duty Cycle**
```cpp
duty = (microseconds / 20000) * 65535
```

ตัวอย่าง:
- 500μs  → (500/20000) × 65535 = 1638
- 1500μs → (1500/20000) × 65535 = 4915
- 2500μs → (2500/20000) × 65535 = 8192

**3. constrain()**
```cpp
microseconds = constrain(microseconds, 500, 2500);
```
จำกัดค่าให้อยู่ในช่วง 500-2500μs ป้องกันค่าผิดพลาด

</details>

## ขั้นตอนที่ 3: ควบคุมด้วยมุม (Angle)

### 3.1 สร้างฟังก์ชันแปลงมุม

```cpp
const int SERVO_PIN = 25;
const int PWM_FREQ = 50;
const int PWM_RESOLUTION = 16;
const int PWM_MAX = 65535;

void setup() {
  Serial.begin(115200);
  ledcAttach(SERVO_PIN, PWM_FREQ, PWM_RESOLUTION);
  
  Serial.println("=== Servo Angle Control ===");
}

int usToDuty(int microseconds) {
  return (microseconds * PWM_MAX) / 20000;
}

void servoWriteUs(int microseconds) {
  microseconds = constrain(microseconds, 500, 2500);
  int duty = usToDuty(microseconds);
  ledcWrite(SERVO_PIN, duty);
}

// ฟังก์ชันควบคุมด้วยมุม
void servoWriteAngle(int angle) {
  // จำกัดมุมให้อยู่ในช่วง 0-180°
  angle = constrain(angle, 0, 180);
  
  // แปลงมุมเป็น microseconds
  // Formula: μs = 500 + (angle × 2000 / 180)
  int microseconds = 500 + (angle * 2000 / 180);
  
  servoWriteUs(microseconds);
  
  Serial.print("Angle: ");
  Serial.print(angle);
  Serial.print("° (");
  Serial.print(microseconds);
  Serial.println(" μs)");
}

void loop() {
  // Sweep จาก 0 ถึง 180
  Serial.println("\n--- Sweep Forward ---");
  for (int angle = 0; angle <= 180; angle += 10) {
    servoWriteAngle(angle);
    delay(200);
  }
  
  delay(500);
  
  // Sweep กลับจาก 180 ถึง 0
  Serial.println("\n--- Sweep Backward ---");
  for (int angle = 180; angle >= 0; angle -= 10) {
    servoWriteAngle(angle);
    delay(200);
  }
  
  delay(500);
}
```

## ขั้นตอนที่ 4: ควบคุมด้วย Serial Command

### 4.1 โปรแกรมควบคุมแบบเรียลไทม์

```cpp
const int SERVO_PIN = 25;
const int PWM_FREQ = 50;
const int PWM_RESOLUTION = 16;
const int PWM_MAX = 65535;

int currentAngle = 90;  // เริ่มต้นที่ 90°

void setup() {
  Serial.begin(115200);
  ledcAttach(SERVO_PIN, PWM_FREQ, PWM_RESOLUTION);
  
  // เริ่มต้นที่ตรงกลาง
  servoWriteAngle(currentAngle);
  
  Serial.println("\n╔═══════════════════════════════╗");
  Serial.println("║  Servo Motor Control Panel   ║");
  Serial.println("╚═══════════════════════════════╝");
  Serial.println("\nCommands:");
  Serial.println("  A[0-180]   - Set angle (e.g., A90)");
  Serial.println("  U[500-2500] - Set pulse in μs (e.g., U1500)");
  Serial.println("  +          - Increase angle by 10°");
  Serial.println("  -          - Decrease angle by 10°");
  Serial.println("  0          - Move to 0°");
  Serial.println("  9          - Move to 90°");
  Serial.println("  1          - Move to 180°");
  Serial.println("  S          - Sweep demo");
  Serial.println("  ?          - Show current position");
  Serial.println("");
}

int usToDuty(int microseconds) {
  return (microseconds * PWM_MAX) / 20000;
}

void servoWriteUs(int microseconds) {
  microseconds = constrain(microseconds, 500, 2500);
  int duty = usToDuty(microseconds);
  ledcWrite(SERVO_PIN, duty);
}

void servoWriteAngle(int angle) {
  angle = constrain(angle, 0, 180);
  currentAngle = angle;
  
  int microseconds = 500 + (angle * 2000 / 180);
  servoWriteUs(microseconds);
  
  Serial.print("✓ Angle: ");
  Serial.print(angle);
  Serial.print("° (");
  Serial.print(microseconds);
  Serial.println(" μs)");
}

void showPosition() {
  int microseconds = 500 + (currentAngle * 2000 / 180);
  Serial.print("Current: ");
  Serial.print(currentAngle);
  Serial.print("° (");
  Serial.print(microseconds);
  Serial.println(" μs)");
}

void sweepDemo() {
  Serial.println("Starting sweep demo...");
  
  for (int angle = 0; angle <= 180; angle += 10) {
    servoWriteAngle(angle);
    delay(100);
  }
  
  delay(300);
  
  for (int angle = 180; angle >= 0; angle -= 10) {
    servoWriteAngle(angle);
    delay(100);
  }
  
  Serial.println("Sweep complete!");
}

void processCommand(String cmd) {
  cmd.trim();
  cmd.toUpperCase();
  
  if (cmd.length() == 0) return;
  
  char firstChar = cmd.charAt(0);
  
  // คำสั่งควบคุมมุม
  if (firstChar == 'A' && cmd.length() > 1) {
    int angle = cmd.substring(1).toInt();
    if (angle >= 0 && angle <= 180) {
      servoWriteAngle(angle);
    } else {
      Serial.println("✗ Error: Angle must be 0-180");
    }
  }
  // คำสั่งควบคุม pulse width (microseconds)
  else if (firstChar == 'U' && cmd.length() > 1) {
    int us = cmd.substring(1).toInt();
    if (us >= 500 && us <= 2500) {
      servoWriteUs(us);
      // คำนวณมุมจาก us
      currentAngle = (us - 500) * 180 / 2000;
      Serial.print("✓ Pulse: ");
      Serial.print(us);
      Serial.print(" μs (≈");
      Serial.print(currentAngle);
      Serial.println("°)");
    } else {
      Serial.println("✗ Error: Pulse must be 500-2500 μs");
    }
  }
  // คำสั่งเพิ่ม/ลดมุม
  else if (cmd == "+") {
    servoWriteAngle(currentAngle + 10);
  }
  else if (cmd == "-") {
    servoWriteAngle(currentAngle - 10);
  }
  // คำสั่งตำแหน่งด่วน
  else if (cmd == "0") {
    servoWriteAngle(0);
  }
  else if (cmd == "9") {
    servoWriteAngle(90);
  }
  else if (cmd == "1") {
    servoWriteAngle(180);
  }
  // คำสั่ง Sweep
  else if (cmd == "S") {
    sweepDemo();
  }
  // แสดงตำแหน่งปัจจุบัน
  else if (cmd == "?") {
    showPosition();
  }
  // คำสั่งไม่รู้จัก
  else {
    Serial.print("✗ Unknown command: ");
    Serial.println(cmd);
  }
}

void loop() {
  if (Serial.available() > 0) {
    String command = Serial.readStringUntil('\n');
    processCommand(command);
  }
}
```

## ขั้นตอนที่ 5: โปรเจคขั้นสูง

### 5.1 ควบคุม Servo หลายตัว

```cpp
const int NUM_SERVOS = 3;
const int SERVO_PINS[] = {25, 26, 27};
int servoAngles[] = {90, 90, 90};  // ตำแหน่งเริ่มต้น

const int PWM_FREQ = 50;
const int PWM_RESOLUTION = 16;
const int PWM_MAX = 65535;

void setup() {
  Serial.begin(115200);
  
  // ตั้งค่า PWM สำหรับแต่ละ servo
  for (int i = 0; i < NUM_SERVOS; i++) {
    ledcAttach(SERVO_PINS[i], PWM_FREQ, PWM_RESOLUTION);
    servoWriteAngle(i, servoAngles[i]);
  }
  
  Serial.println("=== Multi-Servo Control ===");
  Serial.println("Commands: S[0-2] A[0-180]");
  Serial.println("Example: S0 A90 (Servo 0 → 90°)");
}

int usToDuty(int microseconds) {
  return (microseconds * PWM_MAX) / 20000;
}

void servoWriteUs(int servoIndex, int microseconds) {
  microseconds = constrain(microseconds, 500, 2500);
  int duty = usToDuty(microseconds);
  ledcWrite(SERVO_PINS[servoIndex], duty);
}

void servoWriteAngle(int servoIndex, int angle) {
  if (servoIndex < 0 || servoIndex >= NUM_SERVOS) return;
  
  angle = constrain(angle, 0, 180);
  servoAngles[servoIndex] = angle;
  
  int microseconds = 500 + (angle * 2000 / 180);
  servoWriteUs(servoIndex, microseconds);
  
  Serial.print("Servo ");
  Serial.print(servoIndex);
  Serial.print(" → ");
  Serial.print(angle);
  Serial.println("°");
}

void loop() {
  if (Serial.available() > 0) {
    String input = Serial.readStringUntil('\n');
    input.trim();
    input.toUpperCase();
    
    // Format: S0 A90
    int sIndex = input.indexOf('S');
    int aIndex = input.indexOf('A');
    
    if (sIndex >= 0 && aIndex > sIndex) {
      int servoNum = input.substring(sIndex + 1, aIndex).toInt();
      int angle = input.substring(aIndex + 1).toInt();
      
      servoWriteAngle(servoNum, angle);
    }
  }
}
```

### 5.2 Smooth Servo Movement

```cpp
void servoWriteAngleSmooth(int servoIndex, int targetAngle, int delayMs = 15) {
  if (servoIndex < 0 || servoIndex >= NUM_SERVOS) return;
  
  int currentAngle = servoAngles[servoIndex];
  int step = (targetAngle > currentAngle) ? 1 : -1;
  
  for (int angle = currentAngle; angle != targetAngle; angle += step) {
    servoWriteAngle(servoIndex, angle);
    delay(delayMs);
  }
  
  servoWriteAngle(servoIndex, targetAngle);
}
```
