ในบทนี้เราจะเรียนรู้การใช้ Relay Module ควบคุมอุปกรณ์ไฟฟ้า และเชื่อมโยงกับเซ็นเซอร์เพื่อสร้างระบบอัตโนมัติ

## วัตถุประสงค์
- เข้าใจหลักการทำงานของ Relay

## ขั้นตอนที่ 1: รู้จักกับ Relay

### 1.1 Relay คืออะไร?

**Relay** = สวิตช์ไฟฟ้าที่ควบคุมด้วยสัญญาณดิจิทัล

![รูป Relay Module](../assets/images/relay-2ch.png)

**ลักษณะเด่น:**
- ควบคุมไฟแรงดันสูง (AC 220V, DC 30V) ด้วยสัญญาณต่ำ (3.3V/5V)
- แยกวงจรควบคุมและวงจรกำลัง (Isolation)
- มี LED แสดงสถานะการทำงาน
- รองรับกระแสสูง (5A, 10A, 30A ตามรุ่น)

<details markdown="1">
<summary>📖 <b>หลักการทำงานของ Relay</b></summary>

### โครงสร้างภายใน

```
Control Signal (3.3V/5V)
         │
         ▼
    ┌────────┐
    │  Coil  │  ← สร้างสนามแม่เหล็ก
    └────────┘
         │
         ▼
    ┌────────┐
    │ Switch │  ← ดึงให้สัมผัสเปลี่ยน
    └────────┘
         │
         ▼
   High Voltage Circuit
```

### Contact Types

**1. NO (Normally Open) - ปกติเปิด:**
```
สถานะปกติ (Relay OFF):
  COM ○──────X  NO  (ไม่ต่อ)
  
เมื่อ Relay ON:
  COM ●──────●  NO  (ต่อ)
```

**2. NC (Normally Closed) - ปกติปิด:**
```
สถานะปกติ (Relay OFF):
  COM ●──────●  NC  (ต่อ)
  
เมื่อ Relay ON:
  COM ○──────X  NC  (ไม่ต่อ)
```

**3. SPDT (Single Pole Double Throw):**
```
          ┌── NO
  COM ────┤
          └── NC
  
Relay OFF: COM ต่อกับ NC
Relay ON:  COM ต่อกับ NO
```

### ตัวอย่างการใช้งาน

**เปิด-ปิดหลอดไฟ:**
```
AC 220V ───●──────┬── Lamp ─── Neutral
           │      │
        Relay    (NO)
         COM      │
                  └── (ควบคุมด้วย ESP32)
```

</details>

### 1.2 Relay Module 2CH

**คุณสมบัติ:**
- 2 ช่อง (2 Channel) ควบคุมอิสระ
- Optocoupler isolation (ป้องกันไฟกระชาก)
- Active LOW (LOW = Relay ON, HIGH = Relay OFF)
- LED แสดงสถานะแต่ละช่อง
- รองรับ 10A @ 250VAC, 10A @ 30VDC

### 1.3 การต่อวงจร

```
Relay Module 2CH    ESP32
----------------    ------
VCC             →   5V (หรือ external power)
GND             →   GND
IN1             →   GPIO 25 (Relay 1)
IN2             →   GPIO 26 (Relay 2)

Relay Contacts:
NO1, COM1, NC1  →   Load 1 (อุปกรณ์ที่ 1)
NO2, COM2, NC2  →   Load 2 (อุปกรณ์ที่ 2)
```

![รูปต่อวงจร Relay](../assets/images/relay-wiring.png)

⚠️ **คำเตือนความปลอดภัย:**
- **ห้ามต่อไฟบ้าน 220V ถ้าไม่มีความรู้!**
- ทดสอบด้วยแรงดันต่ำก่อน (12V, 24V)
- ใช้ตู้ relay และฉนวนกันไฟฟ้า
- ขอคำปรึกษาช่างไฟฟ้าหากไม่แน่ใจ

## ขั้นตอนที่ 2: โปรแกรมพื้นฐาน

### 2.1 ทดสอบ Relay

```cpp
const int RELAY1_PIN = 25;
const int RELAY2_PIN = 26;

// Relay เป็น Active LOW (LOW = ON)
#define RELAY_ON  LOW
#define RELAY_OFF HIGH

void setup() {
  Serial.begin(115200);
  
  pinMode(RELAY1_PIN, OUTPUT);
  pinMode(RELAY2_PIN, OUTPUT);
  
  // เริ่มต้นปิด relay ทั้งหมด
  digitalWrite(RELAY1_PIN, RELAY_OFF);
  digitalWrite(RELAY2_PIN, RELAY_OFF);
  
  Serial.println("=== Relay Test ===");
  Serial.println("Testing relays...");
}

void loop() {
  // Test Relay 1
  Serial.println("Relay 1 ON");
  digitalWrite(RELAY1_PIN, RELAY_ON);
  delay(2000);
  
  Serial.println("Relay 1 OFF");
  digitalWrite(RELAY1_PIN, RELAY_OFF);
  delay(1000);
  
  // Test Relay 2
  Serial.println("Relay 2 ON");
  digitalWrite(RELAY2_PIN, RELAY_ON);
  delay(2000);
  
  Serial.println("Relay 2 OFF");
  digitalWrite(RELAY2_PIN, RELAY_OFF);
  delay(1000);
  
  // Test Both
  Serial.println("Both Relays ON");
  digitalWrite(RELAY1_PIN, RELAY_ON);
  digitalWrite(RELAY2_PIN, RELAY_ON);
  delay(2000);
  
  Serial.println("Both Relays OFF");
  digitalWrite(RELAY1_PIN, RELAY_OFF);
  digitalWrite(RELAY2_PIN, RELAY_OFF);
  delay(3000);
}
```

### 2.2 ควบคุมด้วย Serial Command

```cpp
const int RELAY1_PIN = 25;
const int RELAY2_PIN = 26;

#define RELAY_ON  LOW
#define RELAY_OFF HIGH

bool relay1State = false;
bool relay2State = false;

void setup() {
  Serial.begin(115200);
  
  pinMode(RELAY1_PIN, OUTPUT);
  pinMode(RELAY2_PIN, OUTPUT);
  
  digitalWrite(RELAY1_PIN, RELAY_OFF);
  digitalWrite(RELAY2_PIN, RELAY_OFF);
  
  Serial.println("\n╔════════════════════════════╗");
  Serial.println("║  Relay Control Panel       ║");
  Serial.println("╚════════════════════════════╝");
  Serial.println("\nCommands:");
  Serial.println("  1ON  / 1OFF  - Relay 1");
  Serial.println("  2ON  / 2OFF  - Relay 2");
  Serial.println("  AON  / AOFF  - All Relays");
  Serial.println("  ?    - Show status");
  Serial.println("");
}

void setRelay(int relayNum, bool state) {
  int pin = (relayNum == 1) ? RELAY1_PIN : RELAY2_PIN;
  digitalWrite(pin, state ? RELAY_ON : RELAY_OFF);
  
  if (relayNum == 1) relay1State = state;
  else relay2State = state;
  
  Serial.print("Relay ");
  Serial.print(relayNum);
  Serial.print(" → ");
  Serial.println(state ? "ON" : "OFF");
}

void showStatus() {
  Serial.println("\n--- Relay Status ---");
  Serial.print("Relay 1: ");
  Serial.println(relay1State ? "ON" : "OFF");
  Serial.print("Relay 2: ");
  Serial.println(relay2State ? "ON" : "OFF");
  Serial.println("-------------------\n");
}

void processCommand(String cmd) {
  cmd.trim();
  cmd.toUpperCase();
  
  if (cmd == "1ON") {
    setRelay(1, true);
  }
  else if (cmd == "1OFF") {
    setRelay(1, false);
  }
  else if (cmd == "2ON") {
    setRelay(2, true);
  }
  else if (cmd == "2OFF") {
    setRelay(2, false);
  }
  else if (cmd == "AON") {
    setRelay(1, true);
    setRelay(2, true);
  }
  else if (cmd == "AOFF") {
    setRelay(1, false);
    setRelay(2, false);
  }
  else if (cmd == "?") {
    showStatus();
  }
  else {
    Serial.print("✗ Unknown command: ");
    Serial.println(cmd);
  }
}

void loop() {
  if (Serial.available() > 0) {
    String command = Serial.readStringUntil('\n');
    processCommand(command);
  }
}
```

## ขั้นตอนที่ 3: ระบบอัตโนมัติ

### 3.1 เปิด-ปิดตามอุณหภูมิ (Auto Fan/Heater)

```cpp
#include <Wire.h>
#include "SparkFunHTU21D.h"

HTU21D htu;

const int RELAY_FAN = 25;     // พัดลม
const int RELAY_HEATER = 26;  // ฮีตเตอร์

#define RELAY_ON  LOW
#define RELAY_OFF HIGH

// Threshold
const float TEMP_HIGH = 30.0;  // เกินนี้เปิดพัดลม
const float TEMP_LOW = 20.0;   // ต่ำกว่านี้เปิดฮีตเตอร์

void setup() {
  Serial.begin(115200);
  Wire.begin();
  
  pinMode(RELAY_FAN, OUTPUT);
  pinMode(RELAY_HEATER, OUTPUT);
  
  digitalWrite(RELAY_FAN, RELAY_OFF);
  digitalWrite(RELAY_HEATER, RELAY_OFF);
  
  if (htu.begin() == false) {
    Serial.println("HTU21D Error!");
    while (1);
  }
  
  Serial.println("Auto Climate Control");
  Serial.println("====================");
  Serial.print("Fan ON if temp > ");
  Serial.print(TEMP_HIGH);
  Serial.println("°C");
  Serial.print("Heater ON if temp < ");
  Serial.print(TEMP_LOW);
  Serial.println("°C");
  Serial.println("");
}

void controlClimate(float temperature) {
  static bool fanState = false;
  static bool heaterState = false;
  
  // ตรวจสอบอุณหภูมิสูง
  if (temperature > TEMP_HIGH && !fanState) {
    digitalWrite(RELAY_FAN, RELAY_ON);
    digitalWrite(RELAY_HEATER, RELAY_OFF);  // ปิดฮีตเตอร์
    fanState = true;
    heaterState = false;
    Serial.println("🌡️  HIGH → Fan ON, Heater OFF");
  }
  // ตรวจสอบอุณหภูมิต่ำ
  else if (temperature < TEMP_LOW && !heaterState) {
    digitalWrite(RELAY_FAN, RELAY_OFF);    // ปิดพัดลม
    digitalWrite(RELAY_HEATER, RELAY_ON);
    fanState = false;
    heaterState = true;
    Serial.println("❄️  LOW → Fan OFF, Heater ON");
  }
  // อุณหภูมิปกติ (Hysteresis: 20-30°C)
  else if (temperature >= TEMP_LOW + 1 && temperature <= TEMP_HIGH - 1) {
    if (fanState || heaterState) {
      digitalWrite(RELAY_FAN, RELAY_OFF);
      digitalWrite(RELAY_HEATER, RELAY_OFF);
      fanState = false;
      heaterState = false;
      Serial.println("✓ NORMAL → All OFF");
    }
  }
}

void loop() {
  float temperature = htu.readTemperature();
  float humidity = htu.readHumidity();
  
  Serial.print("Temp: ");
  Serial.print(temperature, 1);
  Serial.print("°C  |  Humi: ");
  Serial.print(humidity, 1);
  Serial.println("%");
  
  controlClimate(temperature);
  
  delay(5000);  // เช็คทุก 5 วินาที
}
```

### 3.2 ควบคุมตามเวลา (Timer)

```cpp
const int RELAY1_PIN = 25;
const int RELAY2_PIN = 26;

#define RELAY_ON  LOW
#define RELAY_OFF HIGH

struct Schedule {
  int hour;
  int minute;
  int relay;
  bool state;
};

// ตารางเวลา (ตัวอย่าง)
Schedule schedules[] = {
  {8, 0, 1, true},    // 08:00 - Relay 1 ON
  {12, 0, 1, false},  // 12:00 - Relay 1 OFF
  {18, 0, 2, true},   // 18:00 - Relay 2 ON
  {22, 0, 2, false}   // 22:00 - Relay 2 OFF
};
const int NUM_SCHEDULES = 4;

void setup() {
  Serial.begin(115200);
  
  pinMode(RELAY1_PIN, OUTPUT);
  pinMode(RELAY2_PIN, OUTPUT);
  
  digitalWrite(RELAY1_PIN, RELAY_OFF);
  digitalWrite(RELAY2_PIN, RELAY_OFF);
  
  Serial.println("Timer Control System");
  Serial.println("====================");
  printSchedules();
}

void printSchedules() {
  Serial.println("\nSchedule:");
  for (int i = 0; i < NUM_SCHEDULES; i++) {
    Serial.print("  ");
    if (schedules[i].hour < 10) Serial.print("0");
    Serial.print(schedules[i].hour);
    Serial.print(":");
    if (schedules[i].minute < 10) Serial.print("0");
    Serial.print(schedules[i].minute);
    Serial.print(" - Relay ");
    Serial.print(schedules[i].relay);
    Serial.print(" → ");
    Serial.println(schedules[i].state ? "ON" : "OFF");
  }
  Serial.println("");
}

void checkSchedule() {
  // ใช้ uptime แทนเวลาจริง (สำหรับ demo)
  unsigned long totalSeconds = millis() / 1000;
  int hours = (totalSeconds / 3600) % 24;
  int minutes = (totalSeconds % 3600) / 60;
  int seconds = totalSeconds % 60;
  
  // เช็คตารางเวลา
  for (int i = 0; i < NUM_SCHEDULES; i++) {
    if (hours == schedules[i].hour && 
        minutes == schedules[i].minute && 
        seconds == 0) {  // ทำครั้งเดียวต่อนาที
      
      int pin = (schedules[i].relay == 1) ? RELAY1_PIN : RELAY2_PIN;
      digitalWrite(pin, schedules[i].state ? RELAY_ON : RELAY_OFF);
      
      Serial.print("⏰ Schedule triggered: Relay ");
      Serial.print(schedules[i].relay);
      Serial.print(" → ");
      Serial.println(schedules[i].state ? "ON" : "OFF");
    }
  }
}

void loop() {
  checkSchedule();
  delay(1000);  // เช็คทุกวินาที
}
```

## ขั้นตอนที่ 4: โปรเจคขั้นสูง - Smart Home Control

### 4.1 ระบบควบคุมแบบครบวงจร

```cpp
#include <Wire.h>
#include <LiquidCrystal_I2C.h>
#include "SparkFunHTU21D.h"

// Hardware
LiquidCrystal_I2C lcd(0x27, 16, 2);
HTU21D htu;

const int RELAY1_PIN = 25;  // พัดลม/แอร์
const int RELAY2_PIN = 26;  // หลอดไฟ/ปลั๊ก
const int BUTTON_MODE = 18;
const int BUTTON_UP = 19;
const int BUTTON_DOWN = 23;

#define RELAY_ON  LOW
#define RELAY_OFF HIGH

// Modes
enum ControlMode {
  MODE_AUTO,
  MODE_MANUAL,
  MODE_SCHEDULE,
  NUM_MODES
};

ControlMode currentMode = MODE_AUTO;
String modeNames[] = {"AUTO", "MANUAL", "SCHEDULE"};

// Settings
float tempThreshold = 28.0;
bool manualRelay1 = false;
bool manualRelay2 = false;

// Custom character
byte degreeChar[] = {0b00110, 0b01001, 0b01001, 0b00110, 0b00000, 0b00000, 0b00000, 0b00000};

void setup() {
  Serial.begin(115200);
  Wire.begin();
  
  lcd.init();
  lcd.backlight();
  lcd.createChar(0, degreeChar);
  
  pinMode(RELAY1_PIN, OUTPUT);
  pinMode(RELAY2_PIN, OUTPUT);
  pinMode(BUTTON_MODE, INPUT_PULLUP);
  pinMode(BUTTON_UP, INPUT_PULLUP);
  pinMode(BUTTON_DOWN, INPUT_PULLUP);
  
  digitalWrite(RELAY1_PIN, RELAY_OFF);
  digitalWrite(RELAY2_PIN, RELAY_OFF);
  
  if (htu.begin() == false) {
    lcd.print("HTU21D Error!");
    while (1);
  }
  
  lcd.print("Smart Home");
  lcd.setCursor(0, 1);
  lcd.print("Initializing...");
  delay(2000);
  lcd.clear();
}

void handleButtons() {
  static unsigned long lastPress = 0;
  unsigned long now = millis();
  
  if (now - lastPress < 200) return;  // Debounce
  
  // Mode button
  if (digitalRead(BUTTON_MODE) == LOW) {
    currentMode = (ControlMode)((currentMode + 1) % NUM_MODES);
    Serial.print("Mode: ");
    Serial.println(modeNames[currentMode]);
    lastPress = now;
  }
  
  // Up/Down buttons (mode dependent)
  if (currentMode == MODE_AUTO) {
    if (digitalRead(BUTTON_UP) == LOW) {
      tempThreshold += 0.5;
      Serial.print("Threshold: ");
      Serial.println(tempThreshold);
      lastPress = now;
    }
    if (digitalRead(BUTTON_DOWN) == LOW) {
      tempThreshold -= 0.5;
      Serial.print("Threshold: ");
      Serial.println(tempThreshold);
      lastPress = now;
    }
  }
  else if (currentMode == MODE_MANUAL) {
    if (digitalRead(BUTTON_UP) == LOW) {
      manualRelay1 = !manualRelay1;
      digitalWrite(RELAY1_PIN, manualRelay1 ? RELAY_ON : RELAY_OFF);
      Serial.print("Relay 1: ");
      Serial.println(manualRelay1 ? "ON" : "OFF");
      lastPress = now;
    }
    if (digitalRead(BUTTON_DOWN) == LOW) {
      manualRelay2 = !manualRelay2;
      digitalWrite(RELAY2_PIN, manualRelay2 ? RELAY_ON : RELAY_OFF);
      Serial.print("Relay 2: ");
      Serial.println(manualRelay2 ? "ON" : "OFF");
      lastPress = now;
    }
  }
}

void updateAuto(float temperature) {
  static bool relay1State = false;
  
  if (temperature > tempThreshold && !relay1State) {
    digitalWrite(RELAY1_PIN, RELAY_ON);
    relay1State = true;
    Serial.println("Auto: Cooling ON");
  }
  else if (temperature < tempThreshold - 1.0 && relay1State) {
    digitalWrite(RELAY1_PIN, RELAY_OFF);
    relay1State = false;
    Serial.println("Auto: Cooling OFF");
  }
}

void updateDisplay(float temperature, float humidity) {
  // Line 1: Temp, Humi, Mode
  lcd.setCursor(0, 0);
  lcd.print("T:");
  lcd.print(temperature, 1);
  lcd.write(0);  // degree
  lcd.print(" H:");
  lcd.print(humidity, 0);
  lcd.print("% ");
  lcd.print(modeNames[currentMode]);
  lcd.print("  ");
  
  // Line 2: Mode specific
  lcd.setCursor(0, 1);
  if (currentMode == MODE_AUTO) {
    lcd.print("Set:");
    lcd.print(tempThreshold, 1);
    lcd.write(0);
    lcd.print("C       ");
  }
  else if (currentMode == MODE_MANUAL) {
    lcd.print("R1:");
    lcd.print(manualRelay1 ? "ON " : "OFF");
    lcd.print(" R2:");
    lcd.print(manualRelay2 ? "ON " : "OFF");
  }
  else {
    lcd.print("Schedule Mode   ");
  }
}

void loop() {
  static unsigned long lastUpdate = 0;
  unsigned long now = millis();
  
  // Handle buttons
  handleButtons();
  
  // Update every 2 seconds
  if (now - lastUpdate >= 2000) {
    float temperature = htu.readTemperature();
    float humidity = htu.readHumidity();
    
    // Auto mode logic
    if (currentMode == MODE_AUTO) {
      updateAuto(temperature);
    }
    
    // Update display
    updateDisplay(temperature, humidity);
    
    // Serial log
    Serial.print("T: ");
    Serial.print(temperature, 1);
    Serial.print("°C  |  H: ");
    Serial.print(humidity, 1);
    Serial.print("%  |  Mode: ");
    Serial.println(modeNames[currentMode]);
    
    lastUpdate = now;
  }
}
```

## แบบฝึกหัด

### โจทย์ที่ 1: Humidity Control
สร้างระบบควบคุมความชื้นอัตโนมัติ
- เปิดพัดลม/dehumidifier เมื่อความชื้นสูง
- เปิด humidifier เมื่อความชื้นต่ำ

### โจทย์ที่ 2: Multi-Zone Control
ควบคุม 4 relay สำหรับ 4 ห้อง
- แต่ละห้องมี threshold ของตัวเอง
- แสดงสถานะทุกห้องบน LCD (scrolling)

### โจทย์ที่ 3: Energy Saver
บันทึกเวลาที่ relay เปิด/ปิด
- คำนวณการใช้ไฟฟ้า (kWh)
- แสดงสถิติรายวัน/รายสัปดาห์

## ⚠️ ข้อควรระวังด้านความปลอดภัย

### 1. ไฟฟ้าแรงดันสูง
- ❌ **ห้ามทดสอบกับไฟบ้าน 220V ถ้าไม่มีความรู้**
- ✅ ทดสอบด้วยแรงดันต่ำก่อน (12V DC)
- ✅ ใช้ตู้ relay และฉนวนป้องกัน
- ✅ ติดป้ายเตือนชัดเจน

### 2. Load Capacity
- ตรวจสอบกำลังไฟฟ้าของอุปกรณ์ที่ต่อ
- Relay 10A ไม่ควรต่อโหลดเกิน 8A
- ใช้ Contactor สำหรับโหลดขนาดใหญ่

### 3. Wiring
- ใช้สายไฟขนาดเหมาะสม
- ต่อสายให้แน่น ไม่หลวม
- แยกสาย AC และ DC ให้ชัดเจน

### 4. Protection
- ใส่ Fuse หรือ Circuit Breaker
- Ground ให้ถูกต้อง
- ใช้ Surge Protector สำหรับอุปกรณ์สำคัญ
